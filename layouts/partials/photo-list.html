{{- /* 输入一个 Page（相册），返回按拍摄时间倒序的照片列表。 每项是一个 dict： img : 图片资源（可继续 .Resize / .RelPermalink / .Width / .Height） date : 拍摄时间；无 EXIF 时为 1970-01-01 has : 是否真有拍摄时间（false 的在归档页归入「未标注时间」） ym : "2026-08" 形式，归档页分组用 y : "2026" 形式，年份筛选用 ar : 宽高比（宽 / 高），网格布局要用 tags : 该图的标签 = 相册级 tags ∪ 单图级 tags ── 拍摄时间 ── 优先取 EXIF DateTimeOriginal（0x9003）—— 手机原图必然带。 兜底顺序：EXIF → 相册 front matter 的 photoDate → 1970（视为未标注）。 注意：EXIF 里没有时区，存的是「拍摄当时的本地时间」。 所以下面格式化时不做任何时区换算，直接按原值输出， 否则跨时区拍的照片日期会偏移一天。 ── 标签 ── 两级：相册 front matter 的 tags 作用于整个相册； 单图的标签写在 resources 数组里（见 layouts 说明）。 */ -}} {{- $unknown := time.AsTime "1970-01-01" -}} {{- $fallback := .Params.photoDate -}} {{- $albumTags := .Params.tags | default slice -}} {{/* 把 resources 里声明的单图元数据摊平成 文件名 → params 的查找表 */}} {{- $overrides := dict -}} {{- range .Params.resources -}} {{- $overrides = merge $overrides (dict .src (.params | default dict)) -}} {{- end -}} {{- $out := slice -}} {{- range .Resources.ByType "image" -}} {{- $d := $unknown -}} {{- $has := false -}} {{- /* Hugo 0.155+ 的 Image.Meta：日期直接在 .Meta.Date 上。 .Meta.Exif 是个 map（DateTimeOriginal 等原始标签），不是结构体， 写成 .Meta.Exif.Date 会静默取到 nil。 */ -}} {{- with .Meta -}} {{- if not .Date.IsZero -}} {{- $d = .Date -}} {{- $has = true -}} {{- end -}} {{- end -}} {{- if and (not $has) $fallback -}} {{- $d = time.AsTime $fallback -}} {{- $has = true -}} {{- end -}} {{- $own := index $overrides .Name | default dict -}} {{- $tags := union ($own.tags | default slice) $albumTags -}} {{- $out = $out | append (dict "img" . "date" $d "has" $has "ym" ($d.Format "2006-01") "y" ($d.Format "2006") "ar" (div (float .Width) (float .Height)) "tags" $tags ) -}} {{- end -}} {{- return sort $out "date" "desc" -}}
